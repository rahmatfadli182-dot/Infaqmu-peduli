Panduan Setup SQL Supabase

1. Masuk ke dashboard proyek Anda di supabase.com, lalu buka menu SQL Editor → New query.

2. Salin seluruh kode SQL di bawah ini, tempelkan ke editor Supabase, lalu klik tombol Run:

-- 1. Tabel Pengaturan Profil & PIN 6 Digit
CREATE TABLE IF NOT EXISTS infaq_settings (
  id INT PRIMARY KEY,
  org_name TEXT NOT NULL,
  org_handle TEXT NOT NULL,
  org_bio TEXT NOT NULL,
  visitor_pin TEXT NOT NULL,
  admin_pin TEXT NOT NULL
);

INSERT INTO infaq_settings (id, org_name, org_handle, org_bio, visitor_pin, admin_pin)
VALUES (
  1,
  'INFAQMU',
  'infaqmu.peduli',
  'Laporan Transparansi Penerimaan & Penyaluran Infaq Sedekah • Terverifikasi Bendahara',
  '112233',
  '999999'
) ON CONFLICT (id) DO NOTHING;

-- 2. Tabel Transaksi Penerimaan & Penyaluran
CREATE TABLE IF NOT EXISTS infaq_transactions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  type TEXT NOT NULL,
  category TEXT NOT NULL,
  title TEXT NOT NULL,
  description TEXT,
  amount BIGINT NOT NULL,
  tx_date DATE NOT NULL,
  theme TEXT DEFAULT 'emerald',
  aamiin_count INT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Kebijakan Akses Publik API
ALTER TABLE infaq_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE infaq_transactions ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public Access Settings" ON infaq_settings;
CREATE POLICY "Public Access Settings" ON infaq_settings FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Public Access Transactions" ON infaq_transactions;
CREATE POLICY "Public Access Transactions" ON infaq_transactions FOR ALL USING (true) WITH CHECK (true);

3. Setelah berhasil, buka Project Settings → API di Supabase, lalu salin Project URL dan anon public key ke form pengaturan admin.
